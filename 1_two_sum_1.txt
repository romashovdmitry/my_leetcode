// Two Sum: https://leetcode.com/problems/two-sum/


def return_two_sum_index(
    any_array: List[int], target_integer: int
) -> List[int]:

    for index_1 in range(0, len(any_array)):
        index_2_value = target_integer - any_array[index_1]

        for index_2 in range(index_1, len(any_array)):

            if any_array[index_2] == index_2_value:

                return index_1, index_2
